%{
#include <iostream>
#include <string>
#include <cstdlib>
#include "symbol_table.h"
#include "tac.h"
#include "ast_nodes.h"
#include "compiler_context.h"
#include "parser.tab.h"

int comment_depth = 0;
int comment_start_line = 0;
%}

%option noyywrap
%option yylineno

%x COMMENT

DIGIT           [0-9]
INT             {DIGIT}+
DOUBLE          {DIGIT}+\.{DIGIT}+([eE][+-]?{DIGIT}+)?
IDENT           [a-zA-Z_][a-zA-Z0-9_]*
CHAR            \'(\\.|[^\\'\n])\'

%%

[ \t\r]+        { /* Ignore whitespace */ }
\n              { /* yylineno automatically increments */ }

"//".*          { /* Single line comment */ }

"/*"            {
                    comment_depth = 1;
                    comment_start_line = yylineno;
                    BEGIN(COMMENT);
                }

<COMMENT>"/*"   {
                    comment_depth++;
                }

<COMMENT>"*/"   {
                    comment_depth--;
                    if (comment_depth == 0) {
                        BEGIN(INITIAL);
                    }
                }

<COMMENT>\n     { /* newline in comment */ }
<COMMENT>.      { /* comment content */ }

<COMMENT><<EOF>> {
                    g_compiler.reportLexicalError(comment_start_line, "Unterminated multi-line comment starting at line " + std::to_string(comment_start_line));
                    BEGIN(INITIAL);
                }

    /* Keywords */
"let"           { g_compiler.logToken("LET", yytext, yylineno); return LET; }
"var"           { g_compiler.logToken("VAR", yytext, yylineno); return VAR; }
"func"          { g_compiler.logToken("FUNC", yytext, yylineno); return FUNC; }
"return"        { g_compiler.logToken("RETURN", yytext, yylineno); return RETURN; }
"struct"        { g_compiler.logToken("STRUCT", yytext, yylineno); return STRUCT; }
"if"            { g_compiler.logToken("IF", yytext, yylineno); return IF; }
"else"          { g_compiler.logToken("ELSE", yytext, yylineno); return ELSE; }
"switch"        { g_compiler.logToken("SWITCH", yytext, yylineno); return SWITCH; }
"case"          { g_compiler.logToken("CASE", yytext, yylineno); return CASE; }
"default"       { g_compiler.logToken("DEFAULT", yytext, yylineno); return DEFAULT; }
"while"         { g_compiler.logToken("WHILE", yytext, yylineno); return WHILE; }
"repeat"        { g_compiler.logToken("REPEAT", yytext, yylineno); return REPEAT; }
"for"           { g_compiler.logToken("FOR", yytext, yylineno); return FOR; }
"in"            { g_compiler.logToken("IN", yytext, yylineno); return IN; }
"print"         { g_compiler.logToken("PRINT", yytext, yylineno); return PRINT; }

    /* Types */
"Int"           { g_compiler.logToken("TYPE_INT", yytext, yylineno); return TYPE_INT_KW; }
"Double"        { g_compiler.logToken("TYPE_DOUBLE", yytext, yylineno); return TYPE_DOUBLE_KW; }
"Bool"          { g_compiler.logToken("TYPE_BOOL", yytext, yylineno); return TYPE_BOOL_KW; }
"Character"     { g_compiler.logToken("TYPE_CHAR", yytext, yylineno); return TYPE_CHAR_KW; }
"String"        { g_compiler.logToken("TYPE_STRING", yytext, yylineno); return TYPE_STRING_KW; }
"Void"          { g_compiler.logToken("TYPE_VOID", yytext, yylineno); return TYPE_VOID_KW; }

    /* Boolean Literals */
"true"          {
                    yylval.bool_val = 1;
                    g_compiler.logToken("BOOL_LITERAL", yytext, yylineno);
                    return BOOL_LITERAL;
                }
"false"         {
                    yylval.bool_val = 0;
                    g_compiler.logToken("BOOL_LITERAL", yytext, yylineno);
                    return BOOL_LITERAL;
                }

    /* Operators */
"->"            { g_compiler.logToken("ARROW", yytext, yylineno); return ARROW; }
"..."           { g_compiler.logToken("DOTDOTDOT", yytext, yylineno); return DOTDOTDOT; }
"=="            { g_compiler.logToken("EQ", yytext, yylineno); return EQ; }
"!="            { g_compiler.logToken("NE", yytext, yylineno); return NE; }
"<="            { g_compiler.logToken("LE", yytext, yylineno); return LE; }
">="            { g_compiler.logToken("GE", yytext, yylineno); return GE; }
"&&"            { g_compiler.logToken("AND", yytext, yylineno); return AND; }
"||"            { g_compiler.logToken("OR", yytext, yylineno); return OR; }
"!"             { g_compiler.logToken("NOT", yytext, yylineno); return NOT; }

"+"             { g_compiler.logToken("PLUS", yytext, yylineno); return '+'; }
"-"             { g_compiler.logToken("MINUS", yytext, yylineno); return '-'; }
"*"             { g_compiler.logToken("STAR", yytext, yylineno); return '*'; }
"/"             { g_compiler.logToken("SLASH", yytext, yylineno); return '/'; }
"%"             { g_compiler.logToken("PERCENT", yytext, yylineno); return '%'; }
"<"             { g_compiler.logToken("LT", yytext, yylineno); return LT; }
">"             { g_compiler.logToken("GT", yytext, yylineno); return GT; }
"="             { g_compiler.logToken("ASSIGN", yytext, yylineno); return '='; }
"."             { g_compiler.logToken("DOT", yytext, yylineno); return '.'; }

    /* Delimiters */
"("             { g_compiler.logToken("LPAREN", yytext, yylineno); return '('; }
")"             { g_compiler.logToken("RPAREN", yytext, yylineno); return ')'; }
"{"             { g_compiler.logToken("LBRACE", yytext, yylineno); return '{'; }
"}"             { g_compiler.logToken("RBRACE", yytext, yylineno); return '}'; }
"["             { g_compiler.logToken("LBRACKET", yytext, yylineno); return '['; }
"]"             { g_compiler.logToken("RBRACKET", yytext, yylineno); return ']'; }
","             { g_compiler.logToken("COMMA", yytext, yylineno); return ','; }
":"             { g_compiler.logToken("COLON", yytext, yylineno); return ':'; }
";"             { g_compiler.logToken("SEMICOLON", yytext, yylineno); return ';'; }

    /* Literals */
{DOUBLE}        {
                    yylval.double_val = atof(yytext);
                    g_compiler.logToken("DOUBLE_LITERAL", yytext, yylineno);
                    return DOUBLE_LITERAL;
                }

{INT}           {
                    yylval.int_val = atoi(yytext);
                    g_compiler.logToken("INT_LITERAL", yytext, yylineno);
                    return INT_LITERAL;
                }

{CHAR}          {
                    yylval.char_val = yytext[1];
                    g_compiler.logToken("CHAR_LITERAL", yytext, yylineno);
                    return CHAR_LITERAL;
                }

\"(\\.|[^\\"\n])*\" {
                    yylval.str_val = strdup(yytext);
                    g_compiler.logToken("STRING_LITERAL", yytext, yylineno);
                    return STRING_LITERAL;
                }

\"(\\.|[^\\"\n])*   {
                    g_compiler.reportLexicalError(yylineno, "Unterminated string literal: " + std::string(yytext));
                }

    /* Identifiers */
{IDENT}         {
                    yylval.str_val = strdup(yytext);
                    g_compiler.logToken("IDENTIFIER", yytext, yylineno);
                    return IDENTIFIER;
                }

    /* Catch-all for unexpected characters */
.               {
                    g_compiler.reportLexicalError(yylineno, "Unexpected character: '" + std::string(yytext) + "' (ASCII: " + std::to_string((unsigned char)yytext[0]) + ")");
                }

%%
