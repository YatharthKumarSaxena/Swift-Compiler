#ifndef SYMBOL_TABLE_H
#define SYMBOL_TABLE_H

#include <string>
#include <vector>
#include <map>
#include <iostream>
#include <iomanip>

enum DataType {
    TYPE_INT,
    TYPE_DOUBLE,
    TYPE_BOOL,
    TYPE_CHAR,
    TYPE_STRING,
    TYPE_VOID,
    TYPE_ARRAY,
    TYPE_STRUCT,
    TYPE_UNKNOWN,
    TYPE_ERROR
};

std::string dataTypeToString(DataType type);
DataType stringToDataType(const std::string& str);

struct TypeInfo {
    DataType base_type;
    DataType elem_type;        // For arrays
    int array_size;            // For fixed size 1D arrays
    std::string struct_name;   // For struct types

    TypeInfo() : base_type(TYPE_UNKNOWN), elem_type(TYPE_UNKNOWN), array_size(0), struct_name("") {}
    TypeInfo(DataType b) : base_type(b), elem_type(TYPE_UNKNOWN), array_size(0), struct_name("") {}
    TypeInfo(DataType b, DataType elem, int sz) : base_type(b), elem_type(elem), array_size(sz), struct_name("") {}
    TypeInfo(DataType b, const std::string& sname) : base_type(b), elem_type(TYPE_UNKNOWN), array_size(0), struct_name(sname) {}

    bool equals(const TypeInfo& other) const {
        if (base_type != other.base_type) return false;
        if (base_type == TYPE_ARRAY) {
            return elem_type == other.elem_type;
        }
        if (base_type == TYPE_STRUCT) {
            return struct_name == other.struct_name;
        }
        return true;
    }

    std::string toString() const {
        if (base_type == TYPE_ARRAY) {
            return "[" + dataTypeToString(elem_type) + "]";
        }
        if (base_type == TYPE_STRUCT) {
            return struct_name;
        }
        return dataTypeToString(base_type);
    }
};

struct Symbol {
    std::string name;
    TypeInfo type;
    bool is_let;           // true: immutable (let), false: mutable (var)
    bool is_initialized;
    int scope_level;
    int line_declared;

    Symbol() : name(""), is_let(false), is_initialized(false), scope_level(0), line_declared(0) {}
    Symbol(const std::string& n, const TypeInfo& t, bool let_flag, int scope, int line)
        : name(n), type(t), is_let(let_flag), is_initialized(false), scope_level(scope), line_declared(line) {}
};

struct StructMember {
    std::string name;
    TypeInfo type;
    bool is_let;
};

struct StructSymbol {
    std::string name;
    std::vector<StructMember> members;
    int line_declared;

    const StructMember* findMember(const std::string& member_name) const {
        for (const auto& m : members) {
            if (m.name == member_name) return &m;
        }
        return nullptr;
    }
};

struct FuncParam {
    std::string external_label; // External argument label (or same as name)
    std::string internal_name;  // Parameter name
    TypeInfo type;
    bool has_default;
};

struct FunctionSymbol {
    std::string name;
    TypeInfo return_type;
    std::vector<FuncParam> params;
    int line_declared;
};

class SymbolTable {
private:
    std::vector<std::map<std::string, Symbol>> scopes;
    std::map<std::string, StructSymbol> structs;
    std::map<std::string, FunctionSymbol> functions;
    int current_scope_level;

public:
    SymbolTable();

    void enterScope();
    void exitScope();
    int getScopeLevel() const { return current_scope_level; }

    bool insertVar(const std::string& name, const TypeInfo& type, bool is_let, int line);
    Symbol* lookupVar(const std::string& name);
    Symbol* lookupVarCurrentScope(const std::string& name);

    bool insertStruct(const StructSymbol& s);
    StructSymbol* lookupStruct(const std::string& name);

    bool insertFunction(const FunctionSymbol& f);
    FunctionSymbol* lookupFunction(const std::string& name);

    void printTable() const;
};

#endif // SYMBOL_TABLE_H
