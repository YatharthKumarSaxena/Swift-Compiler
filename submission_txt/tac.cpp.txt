#include "tac.h"

std::string Quadruple::toString() const {
    if (op == "label") {
        return result + ":";
    } else if (op == "goto") {
        return "goto " + result;
    } else if (op == "if_true") {
        return "if " + arg1 + " goto " + result;
    } else if (op == "if_false") {
        return "ifFalse " + arg1 + " goto " + result;
    } else if (op == "=") {
        return result + " = " + arg1;
    } else if (op == "array_read") {
        return result + " = " + arg1 + "[" + arg2 + "]";
    } else if (op == "array_write") {
        return arg1 + "[" + arg2 + "] = " + result;
    } else if (op == "field_read") {
        return result + " = " + arg1 + "." + arg2;
    } else if (op == "field_write") {
        return arg1 + "." + arg2 + " = " + result;
    } else if (op == "param") {
        return "param " + arg1;
    } else if (op == "call") {
        if (result.empty()) {
            return "call " + arg1 + ", " + arg2;
        } else {
            return result + " = call " + arg1 + ", " + arg2;
        }
    } else if (op == "return") {
        if (arg1.empty()) {
            return "return";
        } else {
            return "return " + arg1;
        }
    } else if (op == "print") {
        return "print " + arg1;
    } else if (op == "neg") {
        return result + " = -" + arg1;
    } else if (op == "not") {
        return result + " = !" + arg1;
    } else {
        // Binary operations: +, -, *, /, %, ==, !=, <, >, <=, >=, &&, ||
        return result + " = " + arg1 + " " + op + " " + arg2;
    }
}

TACGenerator::TACGenerator() : temp_count(0), label_count(0) {}

std::string TACGenerator::newTemp() {
    return "t" + std::to_string(temp_count++);
}

std::string TACGenerator::newLabel() {
    return "L" + std::to_string(label_count++);
}

void TACGenerator::emit(const std::string& op, const std::string& arg1, const std::string& arg2, const std::string& result) {
    instructions.emplace_back(op, arg1, arg2, result);
}

void TACGenerator::emitLabel(const std::string& label) {
    emit("label", "", "", label);
}

void TACGenerator::emitGoto(const std::string& label) {
    emit("goto", "", "", label);
}

void TACGenerator::emitIfTrue(const std::string& cond, const std::string& label) {
    emit("if_true", cond, "", label);
}

void TACGenerator::emitIfFalse(const std::string& cond, const std::string& label) {
    emit("if_false", cond, "", label);
}

void TACGenerator::emitAssign(const std::string& dest, const std::string& src) {
    emit("=", src, "", dest);
}

void TACGenerator::emitBinary(const std::string& op, const std::string& arg1, const std::string& arg2, const std::string& result) {
    emit(op, arg1, arg2, result);
}

void TACGenerator::emitUnary(const std::string& op, const std::string& arg, const std::string& result) {
    emit(op, arg, "", result);
}

void TACGenerator::emitArrayRead(const std::string& dest, const std::string& array_name, const std::string& index) {
    emit("array_read", array_name, index, dest);
}

void TACGenerator::emitArrayWrite(const std::string& array_name, const std::string& index, const std::string& src) {
    emit("array_write", array_name, index, src);
}

void TACGenerator::emitFieldRead(const std::string& dest, const std::string& struct_var, const std::string& field_name) {
    emit("field_read", struct_var, field_name, dest);
}

void TACGenerator::emitFieldWrite(const std::string& struct_var, const std::string& field_name, const std::string& src) {
    emit("field_write", struct_var, field_name, src);
}

void TACGenerator::emitParam(const std::string& param_name) {
    emit("param", param_name, "", "");
}

void TACGenerator::emitCall(const std::string& func_name, int num_params, const std::string& result) {
    emit("call", func_name, std::to_string(num_params), result);
}

void TACGenerator::emitReturn(const std::string& val) {
    emit("return", val, "", "");
}

void TACGenerator::emitPrint(const std::string& val) {
    emit("print", val, "", "");
}

void TACGenerator::clear() {
    instructions.clear();
    temp_count = 0;
    label_count = 0;
}

void TACGenerator::printTAC() const {
    std::cout << "\n======================== THREE ADDRESS CODE (TAC) ========================\n";
    int idx = 1;
    for (const auto& quad : instructions) {
        if (quad.op == "label") {
            std::cout << std::left << std::setw(6) << (std::to_string(idx++) + ":")
                      << quad.toString() << "\n";
        } else {
            std::cout << std::left << std::setw(6) << (std::to_string(idx++) + ":")
                      << "    " << quad.toString() << "\n";
        }
    }
    std::cout << "==========================================================================\n\n";
}
