#ifndef TAC_H
#define TAC_H

#include <string>
#include <vector>
#include <iostream>
#include <iomanip>

struct Quadruple {
    std::string op;
    std::string arg1;
    std::string arg2;
    std::string result;

    Quadruple(const std::string& o, const std::string& a1, const std::string& a2, const std::string& res)
        : op(o), arg1(a1), arg2(a2), result(res) {}

    std::string toString() const;
};

class TACGenerator {
private:
    std::vector<Quadruple> instructions;
    int temp_count;
    int label_count;

public:
    TACGenerator();

    std::string newTemp();
    std::string newLabel();

    void emit(const std::string& op, const std::string& arg1, const std::string& arg2, const std::string& result);
    void emitLabel(const std::string& label);
    void emitGoto(const std::string& label);
    void emitIfTrue(const std::string& cond, const std::string& label);
    void emitIfFalse(const std::string& cond, const std::string& label);
    void emitAssign(const std::string& dest, const std::string& src);
    void emitBinary(const std::string& op, const std::string& arg1, const std::string& arg2, const std::string& result);
    void emitUnary(const std::string& op, const std::string& arg, const std::string& result);
    void emitArrayRead(const std::string& dest, const std::string& array_name, const std::string& index);
    void emitArrayWrite(const std::string& array_name, const std::string& index, const std::string& src);
    void emitFieldRead(const std::string& dest, const std::string& struct_var, const std::string& field_name);
    void emitFieldWrite(const std::string& struct_var, const std::string& field_name, const std::string& src);
    void emitParam(const std::string& param_name);
    void emitCall(const std::string& func_name, int num_params, const std::string& result);
    void emitReturn(const std::string& val);
    void emitPrint(const std::string& val);

    const std::vector<Quadruple>& getInstructions() const { return instructions; }
    void printTAC() const;
    void clear();
};

#endif // TAC_H
