#include "symbol_table.h"

std::string dataTypeToString(DataType type) {
    switch (type) {
        case TYPE_INT:     return "Int";
        case TYPE_DOUBLE:  return "Double";
        case TYPE_BOOL:    return "Bool";
        case TYPE_CHAR:    return "Character";
        case TYPE_STRING:  return "String";
        case TYPE_VOID:    return "Void";
        case TYPE_ARRAY:   return "Array";
        case TYPE_STRUCT:  return "Struct";
        case TYPE_UNKNOWN: return "Unknown";
        case TYPE_ERROR:   return "TypeError";
        default:           return "Unknown";
    }
}

DataType stringToDataType(const std::string& str) {
    if (str == "Int")       return TYPE_INT;
    if (str == "Double")    return TYPE_DOUBLE;
    if (str == "Bool")      return TYPE_BOOL;
    if (str == "Character") return TYPE_CHAR;
    if (str == "String")    return TYPE_STRING;
    if (str == "Void")      return TYPE_VOID;
    return TYPE_STRUCT; // Custom type or struct
}

SymbolTable::SymbolTable() : current_scope_level(0) {
    // Start with global scope
    scopes.push_back(std::map<std::string, Symbol>());
}

void SymbolTable::enterScope() {
    current_scope_level++;
    scopes.push_back(std::map<std::string, Symbol>());
}

void SymbolTable::exitScope() {
    if (scopes.size() > 1) {
        scopes.pop_back();
        current_scope_level--;
    }
}

bool SymbolTable::insertVar(const std::string& name, const TypeInfo& type, bool is_let, int line) {
    if (scopes.empty()) return false;
    auto& current = scopes.back();
    if (current.find(name) != current.end()) {
        return false; // Redeclaration in same scope
    }
    current[name] = Symbol(name, type, is_let, current_scope_level, line);
    return true;
}

Symbol* SymbolTable::lookupVar(const std::string& name) {
    for (int i = (int)scopes.size() - 1; i >= 0; --i) {
        auto it = scopes[i].find(name);
        if (it != scopes[i].end()) {
            return &(it->second);
        }
    }
    return nullptr;
}

Symbol* SymbolTable::lookupVarCurrentScope(const std::string& name) {
    if (scopes.empty()) return nullptr;
    auto it = scopes.back().find(name);
    if (it != scopes.back().end()) {
        return &(it->second);
    }
    return nullptr;
}

bool SymbolTable::insertStruct(const StructSymbol& s) {
    if (structs.find(s.name) != structs.end()) {
        return false;
    }
    structs[s.name] = s;
    return true;
}

StructSymbol* SymbolTable::lookupStruct(const std::string& name) {
    auto it = structs.find(name);
    if (it != structs.end()) {
        return &(it->second);
    }
    return nullptr;
}

bool SymbolTable::insertFunction(const FunctionSymbol& f) {
    if (functions.find(f.name) != functions.end()) {
        return false;
    }
    functions[f.name] = f;
    return true;
}

FunctionSymbol* SymbolTable::lookupFunction(const std::string& name) {
    auto it = functions.find(name);
    if (it != functions.end()) {
        return &(it->second);
    }
    return nullptr;
}

void SymbolTable::printTable() const {
    std::cout << "\n================================ SYMBOL TABLE ================================\n";
    std::cout << std::left 
              << std::setw(15) << "Identifier"
              << std::setw(12) << "Kind"
              << std::setw(18) << "Type"
              << std::setw(10) << "Scope"
              << std::setw(10) << "Line"
              << "\n";
    std::cout << "------------------------------------------------------------------------------\n";

    for (size_t s = 0; s < scopes.size(); ++s) {
        for (const auto& pair : scopes[s]) {
            const Symbol& sym = pair.second;
            std::cout << std::left 
                      << std::setw(15) << sym.name
                      << std::setw(12) << (sym.is_let ? "let (const)" : "var (mut)")
                      << std::setw(18) << sym.type.toString()
                      << std::setw(10) << sym.scope_level
                      << std::setw(10) << sym.line_declared
                      << "\n";
        }
    }

    if (!functions.empty()) {
        std::cout << "\n--- Functions ---\n";
        for (const auto& pair : functions) {
            const FunctionSymbol& fn = pair.second;
            std::cout << "func " << fn.name << "(";
            for (size_t i = 0; i < fn.params.size(); ++i) {
                if (i > 0) std::cout << ", ";
                std::cout << fn.params[i].internal_name << ": " << fn.params[i].type.toString();
            }
            std::cout << ") -> " << fn.return_type.toString() << " [Line " << fn.line_declared << "]\n";
        }
    }

    if (!structs.empty()) {
        std::cout << "\n--- Structs ---\n";
        for (const auto& pair : structs) {
            const StructSymbol& st = pair.second;
            std::cout << "struct " << st.name << " {\n";
            for (const auto& mem : st.members) {
                std::cout << "    " << (mem.is_let ? "let " : "var ") << mem.name << ": " << mem.type.toString() << "\n";
            }
            std::cout << "} [Line " << st.line_declared << "]\n";
        }
    }
    std::cout << "==============================================================================\n\n";
}
