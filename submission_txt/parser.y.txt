%{

#include <vector>
#include <cstring>

extern int yylex();

extern char* yytext;
void yyerror(const char* s);

CompilerContext g_compiler;

static bool isNumeric(DataType t) {
    return t == TYPE_INT || t == TYPE_DOUBLE;
}

static bool checkTypeCompatibility(const TypeInfo& dest, const TypeInfo& src) {
    if (dest.base_type == TYPE_UNKNOWN || src.base_type == TYPE_UNKNOWN) return true;
    if (dest.base_type == TYPE_ERROR || src.base_type == TYPE_ERROR) return false;
    if (dest.base_type == src.base_type) {
        if (dest.base_type == TYPE_ARRAY) {
            return dest.elem_type == src.elem_type;
        }
        if (dest.base_type == TYPE_STRUCT) {
            return dest.struct_name == src.struct_name;
        }
        return true;
    }
    if (dest.base_type == TYPE_DOUBLE && src.base_type == TYPE_INT) {
        return true;
    }
    return false;
}
%}

%union {
    int int_val;
    double double_val;
    int bool_val;
    char char_val;
    char* str_val;
    struct ExprNode* expr;
    struct ParamNode* param;
    struct ParamList* param_list;
    struct ArgNode* arg;
    struct ArgList* arg_list;
    struct MemberNode* member;
    struct MemberList* member_list;
    struct TypeInfo* type_info;
}

%token LET VAR
%token IF ELSE SWITCH CASE DEFAULT
%token WHILE REPEAT FOR IN
%token FUNC RETURN STRUCT PRINT
%token ARROW DOTDOTDOT

%token TYPE_INT_KW TYPE_DOUBLE_KW TYPE_BOOL_KW TYPE_CHAR_KW TYPE_STRING_KW TYPE_VOID_KW

%token <int_val> INT_LITERAL
%token <double_val> DOUBLE_LITERAL
%token <bool_val> BOOL_LITERAL
%token <char_val> CHAR_LITERAL
%token <str_val> STRING_LITERAL
%token <str_val> IDENTIFIER

%token EQ NE LE GE AND OR NOT

%type <type_info> type_spec
%type <expr> expression literal array_literal call_or_init array_access field_access
%type <param> parameter
%type <param_list> parameters opt_parameters
%type <arg> argument
%type <arg_list> arguments opt_arguments
%type <member> struct_member
%type <member_list> struct_members
%type <str_val> if_header while_header repeat_header

/* Precedence and Associativity */
%right '='
%left OR
%left AND
%left EQ NE
%left LT GT LE GE
%left DOTDOTDOT
%left '+' '-'
%left '*' '/' '%'
%right NOT UMINUS
%left '.' '[' ']' '(' ')'

%start program

%%

program:
      statements
      {
          g_compiler.logParseEvent("Program parsing completed successfully.");
      }
;

statements:
      statements statement
    | /* empty */
;

statement:
      variable_declaration
    | struct_declaration
    | function_declaration
    | assignment opt_semi
    | if_statement
    | switch_statement
    | while_statement
    | repeat_while_statement opt_semi
    | for_in_statement
    | return_statement opt_semi
    | print_statement opt_semi
    | expression opt_semi { delete $1; }
    | ';'
;

opt_semi:
      ';'
    | /* empty */
;

block:
      '{'
      {
          g_compiler.symtab.enterScope();
      }
      statements '}'
      {
          g_compiler.symtab.exitScope();
      }
;

/* ------------------------------------------------------------- */
/* Declarations                                                  */
/* ------------------------------------------------------------- */

variable_declaration:
      /* Type inference: let x = 5 */
      LET IDENTIFIER '=' expression opt_semi
      {
          g_compiler.logParseEvent("Variable declaration (let with inference): " + std::string($2));
          if ($4->type.base_type == TYPE_ERROR) {
              g_compiler.reportSemanticError(yylineno, "Cannot infer type from invalid expression for 'let " + std::string($2) + "'");
          }
          if (!g_compiler.symtab.insertVar($2, $4->type, true, yylineno)) {
              g_compiler.reportSemanticError(yylineno, "Invalid redeclaration of identifier '" + std::string($2) + "' in current scope.");
          } else {
              Symbol* sym = g_compiler.symtab.lookupVar($2);
              if (sym) sym->is_initialized = true;
              g_compiler.tac.emitAssign($2, $4->place);
          }
          free($2);
          delete $4;
      }
    | /* Type inference: var x = 5 */
      VAR IDENTIFIER '=' expression opt_semi
      {
          g_compiler.logParseEvent("Variable declaration (var with inference): " + std::string($2));
          if ($4->type.base_type == TYPE_ERROR) {
              g_compiler.reportSemanticError(yylineno, "Cannot infer type from invalid expression for 'var " + std::string($2) + "'");
          }
          if (!g_compiler.symtab.insertVar($2, $4->type, false, yylineno)) {
              g_compiler.reportSemanticError(yylineno, "Invalid redeclaration of identifier '" + std::string($2) + "' in current scope.");
          } else {
              Symbol* sym = g_compiler.symtab.lookupVar($2);
              if (sym) sym->is_initialized = true;
              g_compiler.tac.emitAssign($2, $4->place);
          }
          free($2);
          delete $4;
      }
    | /* Explicit type: let x: Int = 5 */
      LET IDENTIFIER ':' type_spec '=' expression opt_semi
      {
          g_compiler.logParseEvent("Variable declaration (let with type annotation): " + std::string($2) + ": " + $4->toString());
          if (!checkTypeCompatibility(*$4, $6->type)) {
              g_compiler.reportSemanticError(yylineno, "Cannot assign value of type '" + $6->type.toString() + "' to type '" + $4->toString() + "' in declaration of '" + std::string($2) + "'.");
          }
          if (!g_compiler.symtab.insertVar($2, *$4, true, yylineno)) {
              g_compiler.reportSemanticError(yylineno, "Invalid redeclaration of identifier '" + std::string($2) + "' in current scope.");
          } else {
              Symbol* sym = g_compiler.symtab.lookupVar($2);
              if (sym) sym->is_initialized = true;
              g_compiler.tac.emitAssign($2, $6->place);
          }
          free($2);
          delete $4;
          delete $6;
      }
    | /* Explicit type: var x: Int = 5 */
      VAR IDENTIFIER ':' type_spec '=' expression opt_semi
      {
          g_compiler.logParseEvent("Variable declaration (var with type annotation): " + std::string($2) + ": " + $4->toString());
          if (!checkTypeCompatibility(*$4, $6->type)) {
              g_compiler.reportSemanticError(yylineno, "Cannot assign value of type '" + $6->type.toString() + "' to type '" + $4->toString() + "' in declaration of '" + std::string($2) + "'.");
          }
          if (!g_compiler.symtab.insertVar($2, *$4, false, yylineno)) {
              g_compiler.reportSemanticError(yylineno, "Invalid redeclaration of identifier '" + std::string($2) + "' in current scope.");
          } else {
              Symbol* sym = g_compiler.symtab.lookupVar($2);
              if (sym) sym->is_initialized = true;
              g_compiler.tac.emitAssign($2, $6->place);
          }
          free($2);
          delete $4;
          delete $6;
      }
    | /* Uninitialized var: var x: Int */
      VAR IDENTIFIER ':' type_spec opt_semi
      {
          g_compiler.logParseEvent("Variable declaration (uninitialized var): " + std::string($2) + ": " + $4->toString());
          if (!g_compiler.symtab.insertVar($2, *$4, false, yylineno)) {
              g_compiler.reportSemanticError(yylineno, "Invalid redeclaration of identifier '" + std::string($2) + "' in current scope.");
          }
          free($2);
          delete $4;
      }
    | /* Error: uninitialized let */
      LET IDENTIFIER ':' type_spec opt_semi
      {
          g_compiler.reportSemanticError(yylineno, "Immutable 'let' constant '" + std::string($2) + "' requires an initializing expression.");
          g_compiler.symtab.insertVar($2, *$4, true, yylineno);
          free($2);
          delete $4;
      }
;

type_spec:
      TYPE_INT_KW      { $$ = new TypeInfo(TYPE_INT); }
    | TYPE_DOUBLE_KW   { $$ = new TypeInfo(TYPE_DOUBLE); }
    | TYPE_BOOL_KW     { $$ = new TypeInfo(TYPE_BOOL); }
    | TYPE_CHAR_KW     { $$ = new TypeInfo(TYPE_CHAR); }
    | TYPE_STRING_KW   { $$ = new TypeInfo(TYPE_STRING); }
    | TYPE_VOID_KW     { $$ = new TypeInfo(TYPE_VOID); }
    | '[' type_spec ']'
      {
          $$ = new TypeInfo(TYPE_ARRAY, $2->base_type, 0);
          delete $2;
      }
    | IDENTIFIER
      {
          StructSymbol* st = g_compiler.symtab.lookupStruct($1);
          if (!st) {
              g_compiler.reportSemanticError(yylineno, "Cannot find type '" + std::string($1) + "' in scope.");
              $$ = new TypeInfo(TYPE_ERROR);
          } else {
              $$ = new TypeInfo(TYPE_STRUCT, std::string($1));
          }
          free($1);
      }
;

/* ------------------------------------------------------------- */
/* Structs                                                       */
/* ------------------------------------------------------------- */

struct_declaration:
      STRUCT IDENTIFIER '{' struct_members '}' opt_semi
      {
          g_compiler.logParseEvent("Struct declaration: " + std::string($2));
          StructSymbol st;
          st.name = $2;
          st.line_declared = yylineno;
          for (const auto& m : $4->members) {
              st.members.push_back({m.name, m.type, m.is_let});
          }
          if (!g_compiler.symtab.insertStruct(st)) {
              g_compiler.reportSemanticError(yylineno, "Invalid redeclaration of struct '" + std::string($2) + "'.");
          }
          free($2);
          delete $4;
      }
;

struct_members:
      struct_members struct_member
      {
          $1->members.push_back(*$2);
          delete $2;
          $$ = $1;
      }
    | struct_member
      {
          $$ = new MemberList();
          $$->members.push_back(*$1);
          delete $1;
      }
;

struct_member:
      VAR IDENTIFIER ':' type_spec opt_semi
      {
          $$ = new MemberNode();
          $$->name = $2;
          $$->type = *$4;
          $$->is_let = false;
          free($2);
          delete $4;
      }
    | LET IDENTIFIER ':' type_spec opt_semi
      {
          $$ = new MemberNode();
          $$->name = $2;
          $$->type = *$4;
          $$->is_let = true;
          free($2);
          delete $4;
      }
;

/* ------------------------------------------------------------- */
/* Functions                                                     */
/* ------------------------------------------------------------- */

function_declaration:
      FUNC IDENTIFIER '(' opt_parameters ')' ARROW type_spec
      {
          FunctionSymbol fn;
          fn.name = $2;
          fn.return_type = *$7;
          fn.line_declared = yylineno;
          if ($4) {
              for (const auto& p : $4->params) {
                  fn.params.push_back({p.label, p.name, p.type, false});
              }
          }
          if (!g_compiler.symtab.insertFunction(fn)) {
              g_compiler.reportSemanticError(yylineno, "Invalid redeclaration of function '" + std::string($2) + "'.");
          }
          g_compiler.tac.emitLabel("func_" + std::string($2));
          g_compiler.symtab.enterScope();
          if ($4) {
              for (const auto& p : $4->params) {
                  g_compiler.symtab.insertVar(p.name, p.type, true, yylineno);
              }
          }
      }
      '{' statements '}' opt_semi
      {
          g_compiler.logParseEvent("Function declaration: " + std::string($2) + " -> " + $7->toString());
          g_compiler.symtab.exitScope();
          g_compiler.tac.emit("return", "", "", "");
          free($2);
          delete $7;
          if ($4) delete $4;
      }
    | FUNC IDENTIFIER '(' opt_parameters ')'
      {
          FunctionSymbol fn;
          fn.name = $2;
          fn.return_type = TypeInfo(TYPE_VOID);
          fn.line_declared = yylineno;
          if ($4) {
              for (const auto& p : $4->params) {
                  fn.params.push_back({p.label, p.name, p.type, false});
              }
          }
          if (!g_compiler.symtab.insertFunction(fn)) {
              g_compiler.reportSemanticError(yylineno, "Invalid redeclaration of function '" + std::string($2) + "'.");
          }
          g_compiler.tac.emitLabel("func_" + std::string($2));
          g_compiler.symtab.enterScope();
          if ($4) {
              for (const auto& p : $4->params) {
                  g_compiler.symtab.insertVar(p.name, p.type, true, yylineno);
              }
          }
      }
      '{' statements '}' opt_semi
      {
          g_compiler.logParseEvent("Function declaration (Void): " + std::string($2));
          g_compiler.symtab.exitScope();
          g_compiler.tac.emit("return", "", "", "");
          free($2);
          if ($4) delete $4;
      }
;

opt_parameters:
      parameters  { $$ = $1; }
    | /* empty */ { $$ = nullptr; }
;

parameters:
      parameters ',' parameter
      {
          $1->params.push_back(*$3);
          delete $3;
          $$ = $1;
      }
    | parameter
      {
          $$ = new ParamList();
          $$->params.push_back(*$1);
          delete $1;
      }
;

parameter:
      IDENTIFIER ':' type_spec
      {
          $$ = new ParamNode();
          $$->label = $1;
          $$->name = $1;
          $$->type = *$3;
          free($1);
          delete $3;
      }
    | IDENTIFIER IDENTIFIER ':' type_spec
      {
          $$ = new ParamNode();
          $$->label = $1;
          $$->name = $2;
          $$->type = *$4;
          free($1);
          free($2);
          delete $4;
      }
;

/* ------------------------------------------------------------- */
/* Statements                                                    */
/* ------------------------------------------------------------- */

assignment:
      /* Identifier reassignment: x = 10 */
      IDENTIFIER '=' expression
      {
          g_compiler.logParseEvent("Assignment to variable: " + std::string($1));
          Symbol* sym = g_compiler.symtab.lookupVar($1);
          if (!sym) {
              g_compiler.reportSemanticError(yylineno, "Cannot find identifier '" + std::string($1) + "' in scope.");
          } else {
              if (sym->is_let) {
                  g_compiler.reportSemanticError(yylineno, "Cannot assign to value: '" + std::string($1) + "' is a 'let' constant.");
              }
              if (!checkTypeCompatibility(sym->type, $3->type)) {
                  g_compiler.reportSemanticError(yylineno, "Cannot assign value of type '" + $3->type.toString() + "' to type '" + sym->type.toString() + "'.");
              }
              sym->is_initialized = true;
              g_compiler.tac.emitAssign($1, $3->place);
          }
          free($1);
          delete $3;
      }
    | /* Array element assignment: arr[idx] = 42 */
      IDENTIFIER '[' expression ']' '=' expression
      {
          g_compiler.logParseEvent("Assignment to array element: " + std::string($1) + "[]");
          Symbol* sym = g_compiler.symtab.lookupVar($1);
          if (!sym) {
              g_compiler.reportSemanticError(yylineno, "Cannot find array '" + std::string($1) + "' in scope.");
          } else {
              if (sym->is_let) {
                  g_compiler.reportSemanticError(yylineno, "Cannot mutate elements of immutable 'let' array '" + std::string($1) + "'.");
              }
              if (sym->type.base_type != TYPE_ARRAY) {
                  g_compiler.reportSemanticError(yylineno, "Cannot subscript non-array type '" + sym->type.toString() + "'.");
              }
              if ($3->type.base_type != TYPE_INT) {
                  g_compiler.reportSemanticError(yylineno, "Array index must be of type 'Int', got '" + $3->type.toString() + "'.");
              }
              if (!checkTypeCompatibility(TypeInfo(sym->type.elem_type), $6->type)) {
                  g_compiler.reportSemanticError(yylineno, "Cannot assign value of type '" + $6->type.toString() + "' to array element type '" + dataTypeToString(sym->type.elem_type) + "'.");
              }
              g_compiler.tac.emitArrayWrite($1, $3->place, $6->place);
          }
          free($1);
          delete $3;
          delete $6;
      }
    | /* Struct field assignment: p.x = 10 */
      IDENTIFIER '.' IDENTIFIER '=' expression
      {
          g_compiler.logParseEvent("Assignment to struct member: " + std::string($1) + "." + std::string($3));
          Symbol* sym = g_compiler.symtab.lookupVar($1);
          if (!sym) {
              g_compiler.reportSemanticError(yylineno, "Cannot find identifier '" + std::string($1) + "' in scope.");
          } else {
              if (sym->is_let) {
                  g_compiler.reportSemanticError(yylineno, "Cannot mutate member of immutable 'let' struct '" + std::string($1) + "'.");
              }
              if (sym->type.base_type != TYPE_STRUCT) {
                  g_compiler.reportSemanticError(yylineno, "Type '" + sym->type.toString() + "' is not a struct.");
              } else {
                  StructSymbol* st = g_compiler.symtab.lookupStruct(sym->type.struct_name);
                  if (st) {
                      const StructMember* mem = st->findMember($3);
                      if (!mem) {
                          g_compiler.reportSemanticError(yylineno, "Struct '" + st->name + "' has no member named '" + std::string($3) + "'.");
                      } else {
                          if (mem->is_let) {
                              g_compiler.reportSemanticError(yylineno, "Cannot mutate immutable struct member '" + std::string($3) + "'.");
                          }
                          if (!checkTypeCompatibility(mem->type, $5->type)) {
                              g_compiler.reportSemanticError(yylineno, "Cannot assign value of type '" + $5->type.toString() + "' to member type '" + mem->type.toString() + "'.");
                          }
                      }
                  }
              }
              g_compiler.tac.emitFieldWrite($1, $3, $5->place);
          }
          free($1);
          free($3);
          delete $5;
      }
;

print_statement:
      PRINT '(' expression ')'
      {
          g_compiler.logParseEvent("Print statement: print(" + $3->place + ")");
          g_compiler.tac.emitPrint($3->place);
          delete $3;
      }
;

return_statement:
      RETURN expression
      {
          g_compiler.logParseEvent("Return statement with value: " + $2->place);
          g_compiler.tac.emitReturn($2->place);
          delete $2;
      }
    | RETURN
      {
          g_compiler.logParseEvent("Return statement (void)");
          g_compiler.tac.emitReturn("");
      }
;

/* ------------------------------------------------------------- */
/* Control Flow: If-Else                                         */
/* ------------------------------------------------------------- */

if_header:
      IF expression
      {
          if ($2->type.base_type != TYPE_BOOL && $2->type.base_type != TYPE_ERROR) {
              g_compiler.reportSemanticError(yylineno, "Condition of 'if' statement must be 'Bool', got '" + $2->type.toString() + "'.");
          }
          std::string else_label = g_compiler.tac.newLabel();
          g_compiler.tac.emitIfFalse($2->place, else_label);
          $$ = strdup(else_label.c_str());
          delete $2;
      }
;

if_statement:
      if_header block
      {
          g_compiler.tac.emitLabel($1);
          free($1);
      }
    | if_header block ELSE
      {
          std::string end_label = g_compiler.tac.newLabel();
          g_compiler.tac.emitGoto(end_label);
          g_compiler.tac.emitLabel($1);
          free($1);
          $<str_val>$ = strdup(end_label.c_str());
      }
      block
      {
          g_compiler.tac.emitLabel($<str_val>4);
          free($<str_val>4);
      }
;

/* ------------------------------------------------------------- */
/* Control Flow: Switch                                          */
/* ------------------------------------------------------------- */

switch_statement:
      SWITCH expression '{'
      {
          g_compiler.logParseEvent("Switch statement on expression: " + $2->place);
          g_compiler.enterSwitch($2->place);
          delete $2;
      }
      switch_cases '}'
      {
          g_compiler.exitSwitch();
      }
;

switch_cases:
      switch_cases switch_case
    | switch_case
;

switch_case:
      CASE expression ':'
      {
          std::string next_lbl = g_compiler.startCase($2->place);
          $<str_val>$ = strdup(next_lbl.c_str());
          delete $2;
      }
      statements
      {
          g_compiler.endCase($<str_val>4);
          free($<str_val>4);
      }
    | DEFAULT ':' statements
      {
          g_compiler.logParseEvent("Default case in switch");
      }
;

/* ------------------------------------------------------------- */
/* Loops                                                         */
/* ------------------------------------------------------------- */

while_header:
      WHILE
      {
          std::string loop_start = g_compiler.tac.newLabel();
          g_compiler.tac.emitLabel(loop_start);
          $<str_val>$ = strdup(loop_start.c_str());
      }
      expression
      {
          if ($3->type.base_type != TYPE_BOOL && $3->type.base_type != TYPE_ERROR) {
              g_compiler.reportSemanticError(yylineno, "Condition of 'while' loop must be 'Bool', got '" + $3->type.toString() + "'.");
          }
          std::string loop_end = g_compiler.tac.newLabel();
          g_compiler.tac.emitIfFalse($3->place, loop_end);
          std::string combined = std::string($<str_val>2) + "," + loop_end;
          free($<str_val>2);
          $$ = strdup(combined.c_str());
          delete $3;
      }
;

while_statement:
      while_header block
      {
          std::string s($1);
          size_t comma = s.find(',');
          std::string start_label = s.substr(0, comma);
          std::string end_label = s.substr(comma + 1);
          g_compiler.tac.emitGoto(start_label);
          g_compiler.tac.emitLabel(end_label);
          free($1);
      }
;

repeat_header:
      REPEAT
      {
          std::string loop_start = g_compiler.tac.newLabel();
          g_compiler.tac.emitLabel(loop_start);
          $$ = strdup(loop_start.c_str());
      }
;

repeat_while_statement:
      repeat_header block WHILE expression
      {
          if ($4->type.base_type != TYPE_BOOL && $4->type.base_type != TYPE_ERROR) {
              g_compiler.reportSemanticError(yylineno, "Condition of 'repeat-while' loop must be 'Bool', got '" + $4->type.toString() + "'.");
          }
          g_compiler.tac.emitIfTrue($4->place, $1);
          free($1);
          delete $4;
      }
;

for_in_statement:
      /* Range for loop: for i in 1...10 { ... } */
      FOR IDENTIFIER IN expression DOTDOTDOT expression
      {
          g_compiler.symtab.enterScope();
          g_compiler.symtab.insertVar($2, TypeInfo(TYPE_INT), true, yylineno);
          g_compiler.tac.emitAssign($2, $4->place);
          std::string loop_cond = g_compiler.tac.newLabel();
          std::string loop_end = g_compiler.tac.newLabel();
          g_compiler.tac.emitLabel(loop_cond);
          std::string t = g_compiler.tac.newTemp();
          g_compiler.tac.emitBinary("<=", $2, $6->place, t);
          g_compiler.tac.emitIfFalse(t, loop_end);
          std::string labels = loop_cond + "," + loop_end + "," + std::string($2);
          $<str_val>$ = strdup(labels.c_str());
          delete $4;
          delete $6;
      }
      '{' statements '}'
      {
          std::string s($<str_val>7);
          size_t c1 = s.find(',');
          size_t c2 = s.rfind(',');
          std::string cond_label = s.substr(0, c1);
          std::string end_label = s.substr(c1 + 1, c2 - c1 - 1);
          std::string var_name = s.substr(c2 + 1);

          std::string next_val = g_compiler.tac.newTemp();
          g_compiler.tac.emitBinary("+", var_name, "1", next_val);
          g_compiler.tac.emitAssign(var_name, next_val);
          g_compiler.tac.emitGoto(cond_label);
          g_compiler.tac.emitLabel(end_label);

          g_compiler.symtab.exitScope();
          free($<str_val>7);
          free($2);
      }
    | /* Array for loop: for x in arr { ... } */
      FOR IDENTIFIER IN IDENTIFIER
      {
          g_compiler.symtab.enterScope();
          Symbol* arr_sym = g_compiler.symtab.lookupVar($4);
          TypeInfo elem_t(TYPE_INT);
          int arr_sz = 5;
          if (arr_sym && arr_sym->type.base_type == TYPE_ARRAY) {
              elem_t = TypeInfo(arr_sym->type.elem_type);
              arr_sz = arr_sym->type.array_size > 0 ? arr_sym->type.array_size : 5;
          } else {
              g_compiler.reportSemanticError(yylineno, "Identifier '" + std::string($4) + "' is not an iterable array.");
          }
          g_compiler.symtab.insertVar($2, elem_t, true, yylineno);

          std::string idx_var = g_compiler.tac.newTemp();
          g_compiler.tac.emitAssign(idx_var, "0");
          std::string loop_cond = g_compiler.tac.newLabel();
          std::string loop_end = g_compiler.tac.newLabel();
          g_compiler.tac.emitLabel(loop_cond);
          std::string t_cmp = g_compiler.tac.newTemp();
          g_compiler.tac.emitBinary("<", idx_var, std::to_string(arr_sz), t_cmp);
          g_compiler.tac.emitIfFalse(t_cmp, loop_end);
          g_compiler.tac.emitArrayRead($2, $4, idx_var);

          std::string labels = loop_cond + "," + loop_end + "," + idx_var;
          $<str_val>$ = strdup(labels.c_str());
      }
      '{' statements '}'
      {
          std::string s($<str_val>5);
          size_t c1 = s.find(',');
          size_t c2 = s.rfind(',');
          std::string cond_label = s.substr(0, c1);
          std::string end_label = s.substr(c1 + 1, c2 - c1 - 1);
          std::string idx_var = s.substr(c2 + 1);

          std::string next_idx = g_compiler.tac.newTemp();
          g_compiler.tac.emitBinary("+", idx_var, "1", next_idx);
          g_compiler.tac.emitAssign(idx_var, next_idx);
          g_compiler.tac.emitGoto(cond_label);
          g_compiler.tac.emitLabel(end_label);

          g_compiler.symtab.exitScope();
          free($<str_val>5);
          free($2);
          free($4);
      }
;

/* ------------------------------------------------------------- */
/* Expressions                                                   */
/* ------------------------------------------------------------- */

expression:
      literal
      {
          $$ = $1;
      }
    | IDENTIFIER
      {
          Symbol* sym = g_compiler.symtab.lookupVar($1);
          if (!sym) {
              g_compiler.reportSemanticError(yylineno, "Cannot find identifier '" + std::string($1) + "' in scope.");
              $$ = new ExprNode($1, TypeInfo(TYPE_ERROR));
          } else {
              $$ = new ExprNode($1, sym->type);
              $$->is_lvalue = true;
              $$->is_let = sym->is_let;
          }
          free($1);
      }
    | array_access
      {
          $$ = $1;
      }
    | field_access
      {
          $$ = $1;
      }
    | array_literal
      {
          $$ = $1;
      }
    | call_or_init
      {
          $$ = $1;
      }
    | '(' expression ')'
      {
          $$ = $2;
      }
    | expression '+' expression
      {
          std::string t = g_compiler.tac.newTemp();
          TypeInfo res_type(TYPE_ERROR);
          if ($1->type.base_type == TYPE_STRING && $3->type.base_type == TYPE_STRING) {
              res_type = TypeInfo(TYPE_STRING);
          } else if (isNumeric($1->type.base_type) && isNumeric($3->type.base_type)) {
              res_type = ($1->type.base_type == TYPE_DOUBLE || $3->type.base_type == TYPE_DOUBLE) 
                         ? TypeInfo(TYPE_DOUBLE) : TypeInfo(TYPE_INT);
          } else {
              g_compiler.reportSemanticError(yylineno, "Binary operator '+' cannot be applied to operands of type '" + $1->type.toString() + "' and '" + $3->type.toString() + "'.");
          }
          g_compiler.tac.emitBinary("+", $1->place, $3->place, t);
          $$ = new ExprNode(t, res_type);
          delete $1;
          delete $3;
      }
    | expression '-' expression
      {
          std::string t = g_compiler.tac.newTemp();
          TypeInfo res_type(TYPE_ERROR);
          if (isNumeric($1->type.base_type) && isNumeric($3->type.base_type)) {
              res_type = ($1->type.base_type == TYPE_DOUBLE || $3->type.base_type == TYPE_DOUBLE) 
                         ? TypeInfo(TYPE_DOUBLE) : TypeInfo(TYPE_INT);
          } else {
              g_compiler.reportSemanticError(yylineno, "Binary operator '-' cannot be applied to operands of type '" + $1->type.toString() + "' and '" + $3->type.toString() + "'.");
          }
          g_compiler.tac.emitBinary("-", $1->place, $3->place, t);
          $$ = new ExprNode(t, res_type);
          delete $1;
          delete $3;
      }
    | expression '*' expression
      {
          std::string t = g_compiler.tac.newTemp();
          TypeInfo res_type(TYPE_ERROR);
          if (isNumeric($1->type.base_type) && isNumeric($3->type.base_type)) {
              res_type = ($1->type.base_type == TYPE_DOUBLE || $3->type.base_type == TYPE_DOUBLE) 
                         ? TypeInfo(TYPE_DOUBLE) : TypeInfo(TYPE_INT);
          } else {
              g_compiler.reportSemanticError(yylineno, "Binary operator '*' cannot be applied to operands of type '" + $1->type.toString() + "' and '" + $3->type.toString() + "'.");
          }
          g_compiler.tac.emitBinary("*", $1->place, $3->place, t);
          $$ = new ExprNode(t, res_type);
          delete $1;
          delete $3;
      }
    | expression '/' expression
      {
          std::string t = g_compiler.tac.newTemp();
          TypeInfo res_type(TYPE_ERROR);
          if (isNumeric($1->type.base_type) && isNumeric($3->type.base_type)) {
              res_type = ($1->type.base_type == TYPE_DOUBLE || $3->type.base_type == TYPE_DOUBLE) 
                         ? TypeInfo(TYPE_DOUBLE) : TypeInfo(TYPE_INT);
          } else {
              g_compiler.reportSemanticError(yylineno, "Binary operator '/' cannot be applied to operands of type '" + $1->type.toString() + "' and '" + $3->type.toString() + "'.");
          }
          g_compiler.tac.emitBinary("/", $1->place, $3->place, t);
          $$ = new ExprNode(t, res_type);
          delete $1;
          delete $3;
      }
    | expression '%' expression
      {
          std::string t = g_compiler.tac.newTemp();
          TypeInfo res_type(TYPE_ERROR);
          if ($1->type.base_type == TYPE_INT && $3->type.base_type == TYPE_INT) {
              res_type = TypeInfo(TYPE_INT);
          } else {
              g_compiler.reportSemanticError(yylineno, "Binary operator '%' requires integer operands, got '" + $1->type.toString() + "' and '" + $3->type.toString() + "'.");
          }
          g_compiler.tac.emitBinary("%", $1->place, $3->place, t);
          $$ = new ExprNode(t, res_type);
          delete $1;
          delete $3;
      }
    | expression EQ expression
      {
          std::string t = g_compiler.tac.newTemp();
          if (!checkTypeCompatibility($1->type, $3->type) && !checkTypeCompatibility($3->type, $1->type)) {
              g_compiler.reportSemanticError(yylineno, "Operator '==' cannot compare incompatible types '" + $1->type.toString() + "' and '" + $3->type.toString() + "'.");
          }
          g_compiler.tac.emitBinary("==", $1->place, $3->place, t);
          $$ = new ExprNode(t, TypeInfo(TYPE_BOOL));
          delete $1;
          delete $3;
      }
    | expression NE expression
      {
          std::string t = g_compiler.tac.newTemp();
          if (!checkTypeCompatibility($1->type, $3->type) && !checkTypeCompatibility($3->type, $1->type)) {
              g_compiler.reportSemanticError(yylineno, "Operator '!=' cannot compare incompatible types '" + $1->type.toString() + "' and '" + $3->type.toString() + "'.");
          }
          g_compiler.tac.emitBinary("!=", $1->place, $3->place, t);
          $$ = new ExprNode(t, TypeInfo(TYPE_BOOL));
          delete $1;
          delete $3;
      }
    | expression LT expression
      {
          std::string t = g_compiler.tac.newTemp();
          if (!isNumeric($1->type.base_type) || !isNumeric($3->type.base_type)) {
              g_compiler.reportSemanticError(yylineno, "Relational operator '<' requires numeric operands.");
          }
          g_compiler.tac.emitBinary("<", $1->place, $3->place, t);
          $$ = new ExprNode(t, TypeInfo(TYPE_BOOL));
          delete $1;
          delete $3;
      }
    | expression GT expression
      {
          std::string t = g_compiler.tac.newTemp();
          if (!isNumeric($1->type.base_type) || !isNumeric($3->type.base_type)) {
              g_compiler.reportSemanticError(yylineno, "Relational operator '>' requires numeric operands.");
          }
          g_compiler.tac.emitBinary(">", $1->place, $3->place, t);
          $$ = new ExprNode(t, TypeInfo(TYPE_BOOL));
          delete $1;
          delete $3;
      }
    | expression LE expression
      {
          std::string t = g_compiler.tac.newTemp();
          if (!isNumeric($1->type.base_type) || !isNumeric($3->type.base_type)) {
              g_compiler.reportSemanticError(yylineno, "Relational operator '<=' requires numeric operands.");
          }
          g_compiler.tac.emitBinary("<=", $1->place, $3->place, t);
          $$ = new ExprNode(t, TypeInfo(TYPE_BOOL));
          delete $1;
          delete $3;
      }
    | expression GE expression
      {
          std::string t = g_compiler.tac.newTemp();
          if (!isNumeric($1->type.base_type) || !isNumeric($3->type.base_type)) {
              g_compiler.reportSemanticError(yylineno, "Relational operator '>=' requires numeric operands.");
          }
          g_compiler.tac.emitBinary(">=", $1->place, $3->place, t);
          $$ = new ExprNode(t, TypeInfo(TYPE_BOOL));
          delete $1;
          delete $3;
      }
    | expression AND expression
      {
          std::string t = g_compiler.tac.newTemp();
          if ($1->type.base_type != TYPE_BOOL || $3->type.base_type != TYPE_BOOL) {
              g_compiler.reportSemanticError(yylineno, "Logical operator '&&' requires 'Bool' operands.");
          }
          g_compiler.tac.emitBinary("&&", $1->place, $3->place, t);
          $$ = new ExprNode(t, TypeInfo(TYPE_BOOL));
          delete $1;
          delete $3;
      }
    | expression OR expression
      {
          std::string t = g_compiler.tac.newTemp();
          if ($1->type.base_type != TYPE_BOOL || $3->type.base_type != TYPE_BOOL) {
              g_compiler.reportSemanticError(yylineno, "Logical operator '||' requires 'Bool' operands.");
          }
          g_compiler.tac.emitBinary("||", $1->place, $3->place, t);
          $$ = new ExprNode(t, TypeInfo(TYPE_BOOL));
          delete $1;
          delete $3;
      }
    | NOT expression
      {
          std::string t = g_compiler.tac.newTemp();
          if ($2->type.base_type != TYPE_BOOL) {
              g_compiler.reportSemanticError(yylineno, "Logical operator '!' requires 'Bool' operand.");
          }
          g_compiler.tac.emitUnary("not", $2->place, t);
          $$ = new ExprNode(t, TypeInfo(TYPE_BOOL));
          delete $2;
      }
    | '-' expression %prec UMINUS
      {
          std::string t = g_compiler.tac.newTemp();
          if (!isNumeric($2->type.base_type)) {
              g_compiler.reportSemanticError(yylineno, "Unary operator '-' requires numeric operand.");
          }
          g_compiler.tac.emitUnary("neg", $2->place, t);
          $$ = new ExprNode(t, $2->type);
          delete $2;
      }
;

/* ------------------------------------------------------------- */
/* Literals                                                      */
/* ------------------------------------------------------------- */

literal:
      INT_LITERAL
      {
          $$ = new ExprNode(std::to_string($1), TypeInfo(TYPE_INT));
      }
    | DOUBLE_LITERAL
      {
          $$ = new ExprNode(std::to_string($1), TypeInfo(TYPE_DOUBLE));
      }
    | BOOL_LITERAL
      {
          $$ = new ExprNode($1 ? "true" : "false", TypeInfo(TYPE_BOOL));
      }
    | CHAR_LITERAL
      {
          std::string c = "'";
          c += $1;
          c += "'";
          $$ = new ExprNode(c, TypeInfo(TYPE_CHAR));
      }
    | STRING_LITERAL
      {
          $$ = new ExprNode(std::string($1), TypeInfo(TYPE_STRING));
          free($1);
      }
;

/* Array literal: [1, 2, 3] */
array_literal:
      '[' arguments ']'
      {
          DataType elem_t = TYPE_UNKNOWN;
          if ($2 && !$2->args.empty()) {
              elem_t = $2->args[0].type.base_type;
          }
          int sz = $2 ? (int)$2->args.size() : 0;
          std::string arr_temp = g_compiler.tac.newTemp();
          if ($2) {
              for (int i = 0; i < sz; ++i) {
                  g_compiler.tac.emitArrayWrite(arr_temp, std::to_string(i), $2->args[i].place);
              }
          }
          $$ = new ExprNode(arr_temp, TypeInfo(TYPE_ARRAY, elem_t, sz));
          delete $2;
      }
;

/* Array access: arr[0] */
array_access:
      IDENTIFIER '[' expression ']'
      {
          Symbol* sym = g_compiler.symtab.lookupVar($1);
          TypeInfo elem_t(TYPE_ERROR);
          if (!sym) {
              g_compiler.reportSemanticError(yylineno, "Cannot find array '" + std::string($1) + "' in scope.");
          } else {
              if (sym->type.base_type != TYPE_ARRAY) {
                  g_compiler.reportSemanticError(yylineno, "Cannot subscript non-array type '" + sym->type.toString() + "'.");
              } else {
                  elem_t = TypeInfo(sym->type.elem_type);
              }
              if ($3->type.base_type != TYPE_INT) {
                  g_compiler.reportSemanticError(yylineno, "Array index must be of type 'Int', got '" + $3->type.toString() + "'.");
              }
          }
          std::string t = g_compiler.tac.newTemp();
          g_compiler.tac.emitArrayRead(t, $1, $3->place);
          $$ = new ExprNode(t, elem_t);
          $$->base_id = $1;
          $$->member_id = $3->place;
          $$->is_lvalue = true;
          $$->is_let = sym ? sym->is_let : false;
          free($1);
          delete $3;
      }
;

/* Field access: p.x */
field_access:
      IDENTIFIER '.' IDENTIFIER
      {
          Symbol* sym = g_compiler.symtab.lookupVar($1);
          TypeInfo field_t(TYPE_ERROR);
          bool is_let_field = false;
          if (!sym) {
              g_compiler.reportSemanticError(yylineno, "Cannot find identifier '" + std::string($1) + "' in scope.");
          } else {
              if (sym->type.base_type != TYPE_STRUCT) {
                  g_compiler.reportSemanticError(yylineno, "Cannot access field on non-struct type '" + sym->type.toString() + "'.");
              } else {
                  StructSymbol* st = g_compiler.symtab.lookupStruct(sym->type.struct_name);
                  if (st) {
                      const StructMember* mem = st->findMember($3);
                      if (!mem) {
                          g_compiler.reportSemanticError(yylineno, "Value of type '" + st->name + "' has no member '" + std::string($3) + "'.");
                      } else {
                          field_t = mem->type;
                          is_let_field = mem->is_let;
                      }
                  }
              }
          }
          std::string t = g_compiler.tac.newTemp();
          g_compiler.tac.emitFieldRead(t, $1, $3);
          $$ = new ExprNode(t, field_t);
          $$->base_id = $1;
          $$->member_id = $3;
          $$->is_lvalue = true;
          $$->is_let = sym ? (sym->is_let || is_let_field) : false;
          free($1);
          free($3);
      }
;

/* Struct initialization or function call */
call_or_init:
      IDENTIFIER '(' opt_arguments ')'
      {
          StructSymbol* st = g_compiler.symtab.lookupStruct($1);
          if (st) {
              std::string t = g_compiler.tac.newTemp();
              if ($3) {
                  for (const auto& arg : $3->args) {
                      std::string field_name = arg.label.empty() ? "field" : arg.label;
                      const StructMember* mem = st->findMember(field_name);
                      if (!mem && !arg.label.empty()) {
                          g_compiler.reportSemanticError(yylineno, "Struct '" + st->name + "' has no member named '" + arg.label + "'.");
                      }
                      g_compiler.tac.emitFieldWrite(t, field_name, arg.place);
                  }
              }
              $$ = new ExprNode(t, TypeInfo(TYPE_STRUCT, st->name));
          } else {
              FunctionSymbol* fn = g_compiler.symtab.lookupFunction($1);
              TypeInfo ret_t(TYPE_VOID);
              if (!fn) {
                  g_compiler.reportSemanticError(yylineno, "Cannot find function or struct '" + std::string($1) + "' in scope.");
              } else {
                  ret_t = fn->return_type;
                  int expected_params = (int)fn->params.size();
                  int provided_args = $3 ? (int)$3->args.size() : 0;
                  if (expected_params != provided_args) {
                      g_compiler.reportSemanticError(yylineno, "Function '" + fn->name + "' expects " + std::to_string(expected_params) + " arguments, got " + std::to_string(provided_args) + ".");
                  }
              }
              int num_args = 0;
              if ($3) {
                  num_args = (int)$3->args.size();
                  for (const auto& arg : $3->args) {
                      g_compiler.tac.emitParam(arg.place);
                  }
              }
              std::string res = "";
              if (ret_t.base_type != TYPE_VOID) {
                  res = g_compiler.tac.newTemp();
              }
              g_compiler.tac.emitCall("func_" + std::string($1), num_args, res);
              $$ = new ExprNode(res, ret_t);
          }
          free($1);
          if ($3) delete $3;
      }
;

opt_arguments:
      arguments   { $$ = $1; }
    | /* empty */ { $$ = new ArgList(); }
;

arguments:
      arguments ',' argument
      {
          $1->args.push_back(*$3);
          delete $3;
          $$ = $1;
      }
    | argument
      {
          $$ = new ArgList();
          $$->args.push_back(*$1);
          delete $1;
      }
;

argument:
      IDENTIFIER ':' expression
      {
          $$ = new ArgNode();
          $$->label = $1;
          $$->place = $3->place;
          $$->type = $3->type;
          free($1);
          delete $3;
      }
    | expression
      {
          $$ = new ArgNode();
          $$->label = "";
          $$->place = $1->place;
          $$->type = $1->type;
          delete $1;
      }
;

void yyerror(const char* s) {
    g_compiler.reportSyntaxError(yylineno, std::string(s) + " near '" + yytext + "'");
}
